    rclk,

    rrstn, 
    winc, 
    rinc, 
    wdata, 
    wfull, 
    rempty, 
    rdata

parameter WIDTH = 8;

input wclk;
input rclk;
input wrstn;
input rrstn;
input winc;
input rinc;
input [WIDTH-1:0] wdata;
output wfull;
output rempty;
output [WIDTH-1:0] rdata;

wire [WIDTH-1:0] wdata_reg;
reg [WIDTH-1:0] rdata_reg;
reg [WIDTH-1:0] RAM_MEM [DEPTH-1:0];

// Dual-port RAM module instantiation
dual_port_RAM RAM (
    .wclk(wclk), 
    .wenc(wrstn & winc), 
    .waddr(waddr), 
    .wdata(wdata), 
    .rclk(rclk), 
    .renc(rrstn & rinc), 
    .raddr(raddr), 
    .rdata(rdata_reg)
);

// Write pointer logic
reg [$clog2(DEPTH)-1:0] waddr_bin;
reg [$clog2(DEPTH)-1:0] wptr;
reg [$clog2(DEPTH)-1:0] wptr_buff;

always @(posedge wclk) begin
    if (~wrstn) begin
        waddr_bin <= 0;
    end else if (winc) begin
        waddr_bin <= waddr_bin + 1;
    end
end

always @(posedge wclk) begin
    wptr <= waddr_bin ^ (waddr_bin >> 1);
end

always @(posedge wclk) begin
    if (~wrstn) begin
        wptr_buff <= 0;
    end else begin
        wptr_buff <= wptr;
    end
end

// Read pointer logic

reg [$clog2(DEPTH)-1:0] rptr;
reg [$clog2(DEPTH)-1:0] rptr_buff;
reg [$clog2(DEPTH)-1:0] rptr_syn;

    if (~rrstn) begin
        raddr_bin <= 0;
    end else if (rinc) begin
        raddr_bin <= raddr_bin + 1;
    end
end

always @(posedge rclk) begin
    rptr <= raddr_bin ^ (raddr_bin >> 1);
end

always @(posedge rclk) begin
    if (~rrstn) begin
        rptr_buff <= 0;
    end else begin
        rptr_buff <= rptr;
    end
end

// Two-stage trigger for read pointer synchronization
reg [$clog2(DEPTH)-1:0] rptr_syn_reg;
always @(posedge wclk) begin
    rptr_syn_reg <= rptr;
end

always @(posedge wclk) begin
    rptr_syn <= rptr_syn_reg;
end

// Two-stage trigger for write pointer synchronization
reg [$clog2(DEPTH)-1:0] wptr_syn_reg;
always @(posedge rclk) begin
    wptr_syn_reg <= wptr;
end

reg [$clog2(DEPTH)-1:0] wptr_syn;
always @(posedge rclk) begin
    wptr_syn <= wptr_syn_reg;
end

// Empty and full signals generation
assign wfull = (wptr == {~rptr_syn[$clog2(DEPTH)-1], rptr_syn[$clog2(DEPTH)-2:0]});
assign rempty = (rptr == wptr);

// Address generation
assign waddr = wptr[$clog2(DEPTH)-2:0];
assign raddr = rptr[$clog2(DEPTH)-2:0];

// Output assignments

endmodule

module dual_port_RAM(
    wclk, 
    wenc, 
    waddr, 
    wdata, 
    rclk, 
    renc, 
    raddr, 
    rdata

parameter WIDTH = 8;

input wenc;
input [$clog2(DEPTH)-1:0] waddr;

input rclk;
input renc;

output [WIDTH-1:0] rdata;

reg [WIDTH-1:0] RAM_MEM [DEPTH-1:0];

assign rdata = RAM_MEM[raddr];

always @(posedge wclk) begin
    if (wenc) begin
        RAM_MEM[waddr] <= wdata;
    end
end

endmodule
