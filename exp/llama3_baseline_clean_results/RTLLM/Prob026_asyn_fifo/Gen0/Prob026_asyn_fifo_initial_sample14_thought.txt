module asyn_fifo (
    input wclk, // Write clock signal
    input rclk, // Read clock signal
    input wrstn, // Write reset signal
    input rrstn, // Read reset signal
    input winc, // Write increment signal
    input rinc, // Read increment signal
    input [WIDTH-1:0] wdata, // Write data input
    output wfull, // Write full signal
    output rempty, // Read empty signal
    output [WIDTH-1:0] rdata // Read data output
);

// Define the parameters for the FIFO
parameter WIDTH = 8;
parameter DEPTH = 16;

// Define the dual-port RAM submodule
module dual_port_RAM (
    input wclk,
    input wenc,
    input [$clog2(DEPTH)-1:0] waddr,
    input [WIDTH-1:0] wdata,
    input rclk,
    input renc,
    input [$clog2(DEPTH)-1:0] raddr,
    output reg [WIDTH-1:0] rdata
);

    reg [WIDTH-1:0] RAM_MEM [DEPTH-1:0];

    always @(posedge wclk) begin
        if (wenc) begin
            RAM_MEM[waddr] <= wdata;
        end
    end

    always @(posedge rclk) begin
        if (renc) begin
            rdata <= RAM_MEM[raddr];
        end
    end

endmodule

// Define the Gray code conversion function
function [WIDTH-1:0] gray_encode;
    input [$clog2(DEPTH)-1:0] bin;
    reg [WIDTH-1:0] gray;
    begin
        gray = bin ^ (bin >> 1);
        gray_encode = gray;
    end
endfunction

// Define the Gray code conversion function for 4-bit
function [3:0] gray_encode_4bit;
    input [3:0] bin;
    reg [3:0] gray;
    begin
        gray = bin ^ (bin >> 1);
        gray_encode_4bit = gray;
    end
endfunction

// Define the binary to Gray code conversion for read and write pointers
reg [WIDTH-1:0] wptr, rptr;
reg [$clog2(DEPTH)-1:0] waddr_bin, raddr_bin;
reg [$clog2(DEPTH)-1:0] wptr_bin, rptr_bin;
reg [3:0] wptr_gray, rptr_gray;
reg [3:0] wptr_bin_4bit, rptr_bin_4bit;

// Generate the write and read pointers
always @(posedge wclk) begin
    if (~wrstn) begin
        waddr_bin <= 0;
    end else if (winc) begin
        waddr_bin <= waddr_bin + 1;
    end
end

always @(posedge rclk) begin
    if (~rrstn) begin
        raddr_bin <= 0;
    end else if (rinc) begin
        raddr_bin <= raddr_bin + 1;
    end
end

// Convert write and read pointers to Gray code
assign wptr_bin_4bit = waddr_bin[3:0];
assign wptr_gray = gray_encode_4bit(wptr_bin_4bit);
assign rptr_bin_4bit = raddr_bin[3:0];
assign rptr_gray = gray_encode_4bit(rptr_bin_4bit);

// Instantiate the dual-port RAM submodule
dual_port_RAM RAM (
    .wclk(wclk),
    .wenc(winc),
    .waddr(waddr_bin),
    .wdata(wdata),
    .rclk(rclk),
    .renc(rinc),
    .raddr(raddr_bin),
    .rdata(rdata)
);

// Generate the full and empty signals
reg wfull, rempty;
reg [3:0] rptr_syn;

// Two-stage trigger of the write clock to collect the read pointer
reg [3:0] rptr_sync1, rptr_sync2;
always @(posedge wclk) begin
    rptr_sync1 <= rptr_gray;
    rptr_sync2 <= rptr_sync1;
end
assign rptr_syn = rptr_sync2;

always @(posedge wclk) begin
    if (~wrstn) begin
        wfull <= 0;
        rempty <= 0;
    end else begin
        if (wptr_gray == {~rptr_syn[3], rptr_syn[2:0]}) begin
            wfull <= 1;
        end else begin
            wfull <= 0;
        end
        if (wptr_gray == rptr_syn) begin
            rempty <= 1;
        end else begin
            rempty <= 0;
        end
    end
end

// Assign the output signals
assign wfull = wfull;
assign rempty = rempty;

endmodule
