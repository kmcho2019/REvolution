The module will have two counters, cnt_10 and cnt_100, to keep track of the clock cycles for the 10MHz and 1MHz outputs, respectively. The 50MHz output can be generated by simply inverting the current value of CLK_50 on every positive edge of the input clock. We will use a reset condition to set all output clocks to 0 and reset the counters when the RST signal is active.

module freq_div (
    input CLK_in,
    input RST,
    output reg CLK_50,
    output reg CLK_10,
    output reg CLK_1
);

reg [1:0] cnt_10;
reg [5:0] cnt_100;

always @(posedge CLK_in or posedge RST) begin
    if (RST) begin
        CLK_50 <= 0;
        CLK_10 <= 0;
        CLK_1 <= 0;
        cnt_10 <= 0;
        cnt_100 <= 0;
    end else begin
        // Generate 50MHz clock
        CLK_50 <= ~CLK_50;
        
        // Generate 10MHz clock
        if (cnt_10 == 4) begin
            CLK_10 <= ~CLK_10;
            cnt_10 <= 0;
        end else begin
            cnt_10 <= cnt_10 + 1;
        end
        
        // Generate 1MHz clock
        if (cnt_100 == 49) begin
            CLK_1 <= ~CLK_1;
            cnt_100 <= 0;
        end else begin
            cnt_100 <= cnt_100 + 1;
        end
    end
end

endmodule
