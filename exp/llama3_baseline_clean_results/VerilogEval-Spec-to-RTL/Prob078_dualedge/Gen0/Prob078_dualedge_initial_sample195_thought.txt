One flip-flop will be triggered on the positive edge of the clock and the other on the negative edge. We can use a multiplexer to select the output of the flip-flop that was triggered most recently. module TopModule(
    input clk,
    input d,
    output q
);

reg q_pos, q_neg;
reg prev_clk;

always @(posedge clk) begin
    q_pos <= d;
    prev_clk <= 1'b1;
end

always @(negedge clk) begin
    q_neg <= d;
    prev_clk <= 1'b0;
end

always @(*) begin
    if (prev_clk) begin
        q <= q_pos;
    end else begin
        q <= q_neg;
    end
end

endmodule
